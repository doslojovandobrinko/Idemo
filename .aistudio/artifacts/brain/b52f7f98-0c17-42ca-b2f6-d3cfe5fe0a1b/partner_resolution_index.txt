const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type, x-partner-session",
  "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
};

function jsonResponse(data: unknown, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });
}

async function sha256Hex(data: string): Promise<string> {
  const encoder = new TextEncoder();
  const buffer = await crypto.subtle.digest("SHA-256", encoder.encode(data));
  return Array.from(new Uint8Array(buffer))
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");
}

function validateImageBinarySignature(bytes: Uint8Array, declaredMime: string): { valid: boolean; detectedMime: string | null; error?: string } {
  if (bytes.length < 12) {
    return { valid: false, detectedMime: null, error: "File binary is too small to contain a valid image header." };
  }
  // Check JPEG: FF D8 FF
  if (bytes[0] === 0xff && bytes[1] === 0xd8 && bytes[2] === 0xff) {
    if (declaredMime && declaredMime !== "image/jpeg") {
      return { valid: false, detectedMime: "image/jpeg", error: `Binary signature is JPEG but declared MIME type is ${declaredMime}` };
    }
    return { valid: true, detectedMime: "image/jpeg" };
  }
  // Check PNG: 89 50 4E 47 0D 0A 1A 0A
  if (
    bytes[0] === 0x89 &&
    bytes[1] === 0x50 &&
    bytes[2] === 0x4e &&
    bytes[3] === 0x47 &&
    bytes[4] === 0x0d &&
    bytes[5] === 0x0a &&
    bytes[6] === 0x1a &&
    bytes[7] === 0x0a
  ) {
    if (declaredMime && declaredMime !== "image/png") {
      return { valid: false, detectedMime: "image/png", error: `Binary signature is PNG but declared MIME type is ${declaredMime}` };
    }
    return { valid: true, detectedMime: "image/png" };
  }
  // Check WebP: RIFF (bytes 0..3) + WEBP (bytes 8..11)
  if (
    bytes[0] === 0x52 &&
    bytes[1] === 0x49 &&
    bytes[2] === 0x46 &&
    bytes[3] === 0x46 &&
    bytes[8] === 0x57 &&
    bytes[9] === 0x45 &&
    bytes[10] === 0x42 &&
    bytes[11] === 0x50
  ) {
    if (declaredMime && declaredMime !== "image/webp") {
      return { valid: false, detectedMime: "image/webp", error: `Binary signature is WebP but declared MIME type is ${declaredMime}` };
    }
    return { valid: true, detectedMime: "image/webp" };
  }
  return { valid: false, detectedMime: null, error: "Invalid image file signature or unsupported binary magic bytes." };
}

async function invokeRpc(supabase: any, rpcName: string, params: Record<string, any>) {
  const { data, error } = await supabase.rpc(rpcName, params);
  if (error) {
    return jsonResponse({ success: false, error: "DATABASE_ERROR", message: error.message }, 500);
  }
  return jsonResponse(data, data?.success ? 200 : 400);
}

serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { headers: corsHeaders, status: 204 });
  }

  const url = new URL(req.url);
  const pathname = url.pathname.replace(/\/+$/, "");

  const supabaseUrl = Deno.env.get("SUPABASE_URL") || "";
  const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") || "";

  if (!supabaseUrl || !serviceRoleKey) {
    return jsonResponse({ success: false, error: "SERVER_CONFIG_ERROR", message: "Server database configuration missing." }, 500);
  }

  const supabase = createClient(supabaseUrl, serviceRoleKey);

  try {
    // 1. PUBLIC ENDPOINT: /login
    if (pathname.endsWith("/login") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { public_code, pin } = body;

      if (!public_code || !pin) {
        return jsonResponse({ success: false, error: "MISSING_FIELDS", message: "Partner code and PIN are required." }, 400);
      }

      const rawSessionToken = `idm_pts_${crypto.randomUUID().replace(/-/g, "")}${crypto.randomUUID().replace(/-/g, "")}`;
      const tokenHash = await sha256Hex(rawSessionToken);
      const clientIp = req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "127.0.0.1";
      const userAgent = req.headers.get("user-agent") || "unknown";
      const sourceHash = await sha256Hex(`${clientIp}:${userAgent}`);
      const partnerCodeHash = await sha256Hex(public_code.trim().toLowerCase());
      const expiresAt = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString();

      const { data, error } = await supabase.rpc("authenticate_partner_login", {
        p_public_code: public_code.trim(),
        p_pin: pin.trim(),
        p_token_hash: tokenHash,
        p_expires_at: expiresAt,
        p_source_hash: sourceHash,
        p_partner_code_hash: partnerCodeHash,
      });

      if (error) {
        return jsonResponse({ success: false, error: "AUTH_ERROR", message: error.message }, 500);
      }

      if (!data.success) {
        return jsonResponse({ success: false, error: data.error_code || "INVALID_CREDENTIALS", message: data.message }, 401);
      }

      return jsonResponse({
        success: true,
        session_token: rawSessionToken,
        partner: {
          id: data.partner_id,
          public_code: data.public_code,
          name: data.name,
          must_change_pin: data.must_change_pin,
        },
        expires_at: data.expires_at,
      });
    }

    // 1b. PUBLIC ENDPOINT: /credentials/change-pin (POST)
    if (pathname.endsWith("/credentials/change-pin") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { partner_id, current_pin, new_pin, public_code } = body;

      if (!partner_id || !current_pin || !new_pin) {
        return jsonResponse({
          success: false,
          error: "MISSING_FIELDS",
          message: "partner_id, current_pin, and new_pin are required.",
        }, 400);
      }

      const { data, error } = await supabase.rpc("change_partner_pin_secure", {
        p_partner_id: partner_id,
        p_current_pin: current_pin.trim(),
        p_new_pin: new_pin.trim(),
        p_public_code: typeof public_code === "string" && public_code.trim() !== "" ? public_code.trim() : null,
      });

      if (error) {
        return jsonResponse({ success: false, error: "DATABASE_ERROR", message: error.message }, 500);
      }

      return jsonResponse(data, data?.success ? 200 : 400);
    }

    // 1c. PARTNER AUTHENTICATED ENDPOINT: /profile-content/save (POST)
    if (pathname.endsWith("/profile-content/save") && req.method === "POST") {
      const sessionHeader = req.headers.get("x-partner-session") || req.headers.get("Authorization")?.replace(/^Bearer\s+/i, "") || "";
      if (!sessionHeader) {
        return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Partner authentication session required." }, 401);
      }

      const sessionTokenHash = await sha256Hex(sessionHeader.trim());
      const { data: valData, error: valErr } = await supabase.rpc("validate_partner_session_token", {
        p_token_hash: sessionTokenHash,
      });

      if (valErr || !valData?.valid || !valData?.partner_id) {
        return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Invalid or expired partner session." }, 401);
      }

      const partnerId = valData.partner_id;
      const body = await req.json().catch(() => ({}));

      const {
        introduction,
        photo_base64,
        photo_mime,
        photo_consent,
        contact_phone,
        contact_email,
        canonical_region,
        applied_recs,
        applied_recs_note,
        proposed_rec_title,
        proposed_rec_category,
        proposed_rec_location,
        proposed_rec_rationale,
        partner_type,
        website,
        guide_licence,
        pib,
        languages,
        transport_capacity,
        notice_window,
      } = body;

      let photoPath: string | null = null;
      if (photo_base64 && photo_mime) {
        let binaryStr: string;
        let cleanBase64 = photo_base64;
        if (photo_base64.includes(",")) {
          cleanBase64 = photo_base64.split(",")[1];
        }
        try {
          binaryStr = atob(cleanBase64);
        } catch {
          return jsonResponse({ success: false, error: "INVALID_PHOTO_ENCODING", message: "Failed to decode base64 photo data." }, 400);
        }

        const bytes = new Uint8Array(binaryStr.length);
        for (let i = 0; i < binaryStr.length; i++) {
          bytes[i] = binaryStr.charCodeAt(i);
        }

        const sigResult = validateImageBinarySignature(bytes, photo_mime);
        if (!sigResult.valid) {
          return jsonResponse({ success: false, error: "INVALID_PHOTO_SIGNATURE", message: sigResult.error || "Invalid image binary header signature." }, 400);
        }

        const ext = sigResult.detectedMime === "image/png" ? "png" : sigResult.detectedMime === "image/webp" ? "webp" : "jpg";
        const filename = `${partnerId}_draft_${Date.now()}.${ext}`;
        photoPath = `${partnerId}/${filename}`;

        const { error: uploadErr } = await supabase.storage
          .from("partner-passports")
          .upload(photoPath, bytes, {
            contentType: sigResult.detectedMime || photo_mime,
            upsert: true,
          });

        if (uploadErr) {
          return jsonResponse({ success: false, error: "STORAGE_UPLOAD_FAILED", message: `Failed to upload photo asset: ${uploadErr.message}` }, 500);
        }
      }

      return invokeRpc(supabase, "save_partner_profile_draft_secure", {
        p_partner_id: partnerId,
        p_intro_draft: typeof introduction === "string" ? introduction.trim() : null,
        p_draft_photo_path: photoPath,
        p_draft_photo_mime: photo_mime || null,
        p_photo_consent: Boolean(photo_consent),
        p_contact_phone: typeof contact_phone === "string" ? contact_phone.trim() : null,
        p_contact_email: typeof contact_email === "string" ? contact_email.trim() : null,
        p_canonical_region: typeof canonical_region === "string" ? canonical_region.trim() : null,
        p_applied_recs: Array.isArray(applied_recs) ? applied_recs : null,
        p_applied_recs_note: typeof applied_recs_note === "string" ? applied_recs_note.trim() : null,
        p_proposed_rec_title: typeof proposed_rec_title === "string" ? proposed_rec_title.trim() : null,
        p_proposed_rec_category: typeof proposed_rec_category === "string" ? proposed_rec_category.trim() : null,
        p_proposed_rec_location: typeof proposed_rec_location === "string" ? proposed_rec_location.trim() : null,
        p_proposed_rec_rationale: typeof proposed_rec_rationale === "string" ? proposed_rec_rationale.trim() : null,
        p_partner_type: typeof partner_type === "string" ? partner_type.trim() : null,
        p_website: typeof website === "string" ? website.trim() : null,
        p_guide_licence: typeof guide_licence === "string" ? guide_licence.trim() : null,
        p_pib: typeof pib === "string" ? pib.trim() : null,
        p_languages: Array.isArray(languages) ? languages : null,
        p_transport_capacity: typeof transport_capacity === "string" ? transport_capacity.trim() : null,
        p_notice_window: typeof notice_window === "string" ? notice_window.trim() : null,
      });
    }

    // 1d. PARTNER AUTHENTICATED ENDPOINT: /profile-content/submit (POST)
    if (pathname.endsWith("/profile-content/submit") && req.method === "POST") {
      const sessionHeader = req.headers.get("x-partner-session") || req.headers.get("Authorization")?.replace(/^Bearer\s+/i, "") || "";
      if (!sessionHeader) {
        return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Partner authentication session required." }, 401);
      }

      const sessionTokenHash = await sha256Hex(sessionHeader.trim());
      const { data: valData, error: valErr } = await supabase.rpc("validate_partner_session_token", {
        p_token_hash: sessionTokenHash,
      });

      if (valErr || !valData?.valid || !valData?.partner_id) {
        return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Invalid or expired partner session." }, 401);
      }

      return invokeRpc(supabase, "submit_partner_profile_secure", {
        p_partner_id: valData.partner_id,
      });
    }

    // 2. STUDIO / ADMIN ENDPOINT: /admin/profile-review (POST)
    if (pathname.endsWith("/admin/profile-review") && req.method === "POST") {
      if (req.headers.has("x-partner-session") && !req.headers.has("authorization")) {
        return jsonResponse({
          success: false,
          error: "UNAUTHORIZED",
          message: "Partner sessions are not authorized to perform administrative or editorial profile reviews.",
        }, 403);
      }

      const authHeader = req.headers.get("Authorization");
      if (!authHeader || !authHeader.startsWith("Bearer ")) {
        return jsonResponse({
          success: false,
          error: "UNAUTHORIZED",
          message: "Studio/admin authentication token is required in Authorization header.",
        }, 401);
      }

      const token = authHeader.substring(7).trim();
      const { data: { user }, error: authErr } = await supabase.auth.getUser(token);
      if (authErr || !user) {
        return jsonResponse({
          success: false,
          error: "UNAUTHORIZED",
          message: "Invalid or expired Studio/admin authentication session.",
        }, 401);
      }

      const reviewerId = user.id;
      const rawRole1 = user.app_metadata?.role || user.role || "";
      const roleStr1 = Array.isArray(rawRole1) ? rawRole1[0] : String(rawRole1);
      const userRole1 = roleStr1.toLowerCase().replace(/[\s_-]+/g, "");
      const allowedRoles = ["editoriallead", "superadmin", "curator", "partnermanager"];
      const isAllowedRole1 = allowedRoles.includes(userRole1);

      if (!isAllowedRole1) {
        return jsonResponse({
          success: false,
          error: "FORBIDDEN",
          message: "Insufficient permissions. Only authorized Studio roles can review partner profiles.",
        }, 403);
      }

      const body = await req.json().catch(() => ({}));
      const { partner_id, action, review_note } = body;

      if (!partner_id || !action) {
        return jsonResponse({ success: false, error: "MISSING_FIELDS", message: "partner_id and action are required." }, 400);
      }

      return invokeRpc(supabase, "review_partner_profile_secure", {
        p_partner_id: partner_id,
        p_reviewer_id: reviewerId,
        p_action: action,
        p_review_note: review_note || null,
      });
    }

    // 2b. STUDIO / ADMIN ENDPOINT: /admin/profile-queue (GET)
    if (pathname.endsWith("/admin/profile-queue") && req.method === "GET") {
      if (req.headers.has("x-partner-session") && !req.headers.has("authorization")) {
        return jsonResponse({
          success: false,
          error: "UNAUTHORIZED",
          message: "Partner sessions are not authorized to perform administrative or editorial profile reviews.",
        }, 403);
      }

      const authHeader = req.headers.get("Authorization");
      if (!authHeader || !authHeader.startsWith("Bearer ")) {
        return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Valid Studio authentication is required." }, 401);
      }

      const token = authHeader.substring(7).trim();
      const { data: { user }, error: authErr } = await supabase.auth.getUser(token);
      if (authErr || !user) {
        return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Valid Studio authentication is required." }, 401);
      }

      const rawRole2 = user.app_metadata?.role || user.role || "";
      const roleStr2 = Array.isArray(rawRole2) ? rawRole2[0] : String(rawRole2);
      const userRole2 = roleStr2.toLowerCase().replace(/[\s_-]+/g, "");
      const allowedRoles = ["editoriallead", "superadmin", "curator", "partnermanager"];
      const isAllowedRole2 = allowedRoles.includes(userRole2);

      if (!isAllowedRole2) {
        return jsonResponse({ success: false, error: "FORBIDDEN", message: "Editorial review access is restricted." }, 403);
      }

      const statusParam = (url.searchParams.get("status") || "pending_review").toLowerCase().trim();
      const allowedStatuses = ["pending_review", "changes_requested", "approved", "all"];
      if (!allowedStatuses.includes(statusParam)) {
        return jsonResponse({ success: false, error: "INVALID_STATUS_FILTER", message: "Unsupported Partner Passport review status filter." }, 400);
      }

      let query = supabase
        .from("partner_profile_content")
        .select(`
          partner_id, intro_draft, intro_published, draft_photo_path, published_photo_path,
          draft_photo_mime, published_photo_mime, review_status, photo_consent_given,
          photo_consent_at, photo_consent_withdrawn_at, submitted_at, reviewed_at, reviewed_by,
          review_note, content_version, created_at, updated_at,
          draft_contact_phone, draft_contact_email, published_contact_phone, published_contact_email,
          canonical_region, applied_recs, applied_recs_note,
          proposed_rec_title, proposed_rec_category, proposed_rec_location, proposed_rec_rationale,
          partner_type, website, guide_licence, pib, languages, transport_capacity, notice_window,
          partners!inner ( id, public_code, name, status )
        `);

      if (statusParam === "all") {
        query = query.in("review_status", ["pending_review", "changes_requested", "approved"]);
      } else {
        query = query.eq("review_status", statusParam);
      }

      const { data: records, error: dbErr } = await query;
      if (dbErr) {
        return jsonResponse({ success: false, error: "DATABASE_ERROR", message: "Failed to fetch partner profile review queue." }, 500);
      }

      const statusPriority: Record<string, number> = { pending_review: 1, changes_requested: 2, approved: 3 };

      const profiles = await Promise.all(
        (records || []).map(async (row: any) => {
          const partner = row.partners || {};
          const isApproved = row.review_status === "approved";
          const photoPath = isApproved ? row.published_photo_path : row.draft_photo_path;
          const isConsentGiven = row.photo_consent_given === true;
          const isConsentNotWithdrawn = !row.photo_consent_withdrawn_at;
          const hasPhotoPath = typeof photoPath === "string" && photoPath.trim().length > 0;

          let photoAvailable = false;
          let photoUrl: string | null = null;

          if (isConsentGiven && isConsentNotWithdrawn && hasPhotoPath) {
            const rawPhotoPath = photoPath.trim();
            if (
              rawPhotoPath.startsWith("/") ||
              rawPhotoPath.startsWith("http://") ||
              rawPhotoPath.startsWith("https://") ||
              rawPhotoPath.startsWith("data:")
            ) {
              photoAvailable = true;
              photoUrl = rawPhotoPath;
            } else {
              try {
                const { data: signedData, error: signedErr } = await supabase.storage
                  .from("partner-passports")
                  .createSignedUrl(rawPhotoPath, 300);
                if (!signedErr && signedData?.signedUrl) {
                  photoAvailable = true;
                  photoUrl = signedData.signedUrl;
                }
              } catch {
                photoAvailable = false;
                photoUrl = null;
              }
            }
          }

          const rawIntro = row.intro_draft || row.intro_published || "";
          const cleanIntro = rawIntro.trim();
          const introWordCount = cleanIntro ? cleanIntro.split(/\s+/).filter(Boolean).length : 0;

          return {
            partner_id: row.partner_id,
            partner_code: partner.public_code || "",
            partner_name: partner.name || "",
            partner_status: partner.status || "",
            partner_type: row.partner_type || "Individual",
            website: row.website || null,
            guide_licence: row.guide_licence || null,
            pib: row.pib || null,
            languages: Array.isArray(row.languages) ? row.languages : [],
            transport_capacity: row.transport_capacity || "none",
            notice_window: row.notice_window || "24h",
            review_status: row.review_status,
            introduction_draft: row.intro_draft || null,
            introduction_published: row.intro_published || null,
            introduction_word_count: introWordCount,
            photo_consent_given: isConsentGiven,
            photo_consent_withdrawn: !isConsentNotWithdrawn,
            photo_available: photoAvailable,
            photo_url: photoUrl,
            draft_contact_phone: row.draft_contact_phone || null,
            draft_contact_email: row.draft_contact_email || null,
            published_contact_phone: row.published_contact_phone || null,
            published_contact_email: row.published_contact_email || null,
            canonical_region: row.canonical_region || null,
            applied_recs: row.applied_recs || null,
            applied_recs_note: row.applied_recs_note || null,
            proposed_rec_title: row.proposed_rec_title || null,
            proposed_rec_category: row.proposed_rec_category || null,
            proposed_rec_location: row.proposed_rec_location || null,
            proposed_rec_rationale: row.proposed_rec_rationale || null,
            submitted_at: row.submitted_at || null,
            reviewed_at: row.reviewed_at || null,
            reviewer_note: row.review_note || null,
            content_version: row.content_version || 1,
            updated_at: row.updated_at || null,
          };
        })
      );

      profiles.sort((a, b) => {
        const pA = statusPriority[a.review_status] || 99;
        const pB = statusPriority[b.review_status] || 99;
        if (pA !== pB) return pA - pB;
        const tA = a.submitted_at ? new Date(a.submitted_at).getTime() : 0;
        const tB = b.submitted_at ? new Date(b.submitted_at).getTime() : 0;
        return tB - tA;
      });

      return jsonResponse({
        success: true,
        status_filter: statusParam,
        count: profiles.length,
        profiles,
      });
    }

    // 3. PARTNER OPPORTUNITIES ENDPOINTS
    const sessionHeader = req.headers.get("x-partner-session") || req.headers.get("Authorization")?.replace(/^Bearer\s+/i, "") || "";
    if (!sessionHeader) {
      return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Partner session is required for opportunity resolution." }, 401);
    }

    const sessionTokenHash = await sha256Hex(sessionHeader.trim());
    const { data: valData, error: valErr } = await supabase.rpc("validate_partner_session_token", {
      p_token_hash: sessionTokenHash,
    });

    if (valErr || !valData?.valid || !valData?.partner_id) {
      return jsonResponse({ success: false, error: "UNAUTHORIZED", message: "Invalid or expired partner session token." }, 401);
    }

    const partnerId = valData.partner_id;

    // Helper: Push notification dispatch for visitor proposals
    const dispatchVisitorProposalPushByMatch = async (inquiryMatchId: string, resolvedResponseId: string) => {
      try {
        const oneSignalAppId = Deno.env.get("ONESIGNAL_APP_ID") || "";
        const oneSignalApiKey = Deno.env.get("ONESIGNAL_REST_API_KEY") || "";

        const { data: dispatchCtx, error: ctxErr } = await supabase.rpc("get_visitor_proposal_push_context", {
          p_match_id: inquiryMatchId,
        });

        if (ctxErr || !dispatchCtx || !dispatchCtx.success) {
          const errMsg = ctxErr?.message || dispatchCtx?.message || "Failed to resolve visitor proposal push context.";
          return { sent: false, error: errMsg };
        }

        const inquiryId = dispatchCtx.inquiry_id;
        const targetOutboxKey = `push_visitor_proposal_${resolvedResponseId}`;

        const { data: job, error: claimErr } = await supabase.rpc("claim_notification_outbox_job", {
          p_channel: "push_visitor",
          p_idempotency_key: targetOutboxKey,
          p_recipient_id: dispatchCtx.visitor_user_id || "00000000-0000-0000-0000-000000000000",
          p_payload: {
            inquiry_id: inquiryId,
            match_id: inquiryMatchId,
            response_id: resolvedResponseId,
            action: "review_proposal",
          },
        });

        if (claimErr || !job) {
          const errMsg = claimErr?.message || "Job already processed or locked by another worker.";
          return { sent: false, duplicate_skipped: true, error: errMsg };
        }

        const recordOutboxFailure = async (errCode: string, errDetails: string) => {
          if (job?.id) {
            await supabase
              .from("notification_outbox")
              .update({
                status: "failed",
                error_code: errCode,
                error_details: errDetails,
                updated_at: new Date().toISOString(),
              })
              .eq("id", job.id);
          }
        };

        if (!oneSignalAppId || !oneSignalApiKey) {
          const errMsg = "ONESIGNAL_CONFIG_MISSING: Notification secrets missing or unconfigured in backend secrets.";
          await recordOutboxFailure("MISSING_ONESIGNAL_CONFIG", errMsg);
          return { sent: false, error: errMsg };
        }

        const visitorExternalId =
          typeof dispatchCtx.visitor_external_id === "string" && dispatchCtx.visitor_external_id.trim() !== ""
            ? dispatchCtx.visitor_external_id.trim()
            : "";

        if (!visitorExternalId) {
          const errMsg = "MISSING_VISITOR_PUSH_IDENTITY: Inquiry has no bound OneSignal external_id.";
          await recordOutboxFailure("MISSING_VISITOR_PUSH_IDENTITY", errMsg);
          return { sent: false, error: errMsg };
        }

        const unreadCount = Math.max(0, Number(dispatchCtx.unread_count ?? 1));

        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 5000);

        try {
          const osResponse = await fetch("https://api.onesignal.com/notifications", {
            method: "POST",
            headers: {
              "Content-Type": "application/json; charset=utf-8",
              "Authorization": `Key ${oneSignalApiKey}`,
            },
            body: JSON.stringify({
              app_id: oneSignalAppId,
              target_channel: "push",
              include_aliases: {
                external_id: [visitorExternalId],
              },
              headings: {
                en: "IDEMO",
              },
              contents: {
                en: "You have a new partner proposal ready to review.",
              },
              ios_badgeType: "SetTo",
              ios_badgeCount: unreadCount,
              idempotency_key: resolvedResponseId,
              data: {
                inquiry_id: inquiryId,
                match_id: inquiryMatchId,
                response_id: resolvedResponseId,
                action: "review_proposal",
              },
            }),
            signal: controller.signal,
          });

          const osBody = await osResponse.json().catch(() => ({}));

          if (!osResponse.ok || !osBody?.id) {
            const errDetail = osBody?.errors ? JSON.stringify(osBody.errors) : `HTTP_${osResponse.status}`;
            const errMsg = `ONESIGNAL_DISPATCH_FAILED: ${errDetail}`;
            await recordOutboxFailure("ONESIGNAL_DISPATCH_FAILED", errMsg);
            return { sent: false, error: errMsg };
          }

          const providerMessageId = String(osBody.id);

          if (job?.id) {
            await supabase
              .from("notification_outbox")
              .update({
                status: "sent",
                idempotency_key: targetOutboxKey,
                provider_message_id: providerMessageId,
                processed_at: new Date().toISOString(),
                updated_at: new Date().toISOString(),
              })
              .eq("id", job.id);
          }

          return { sent: true, provider_message_id: providerMessageId, badge_count: unreadCount };
        } catch (fetchErr: any) {
          const isTimeout = fetchErr?.name === "AbortError";
          const errMsg = isTimeout
            ? "ONESIGNAL_TIMEOUT: OneSignal API call timed out after 5000ms."
            : `ONESIGNAL_FETCH_ERROR: ${fetchErr?.message || "Unknown fetch error"}`;
          await recordOutboxFailure(isTimeout ? "ONESIGNAL_TIMEOUT" : "ONESIGNAL_FETCH_ERROR", errMsg);
          return { sent: false, error: errMsg };
        } finally {
          clearTimeout(timeoutId);
        }
      } catch (pushErr: any) {
        console.error("[PARTNER_RESOLUTION_PUSH_ERROR]", pushErr?.message || pushErr);
        return { sent: false, error: pushErr?.message || "Unknown push dispatch error" };
      }
    };

    // ENDPOINT: /opportunities/accept (POST)
    if (pathname.endsWith("/opportunities/accept") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { match_id, message } = body;

      if (!match_id || !message) {
        return jsonResponse({ success: false, error: "MISSING_FIELDS", message: "match_id and message are required." }, 400);
      }

      const { data, error } = await supabase.rpc("accept_partner_opportunity_secure", {
        p_partner_id: partnerId,
        p_match_id: match_id,
        p_message: message,
      });

      if (error) {
        return jsonResponse({ success: false, error: "DATABASE_ERROR", message: error.message }, 500);
      }

      if (!data?.success) {
        return jsonResponse(data, 400);
      }

      const pushResult = await dispatchVisitorProposalPushByMatch(match_id, data.response_id);
      return jsonResponse({ ...data, push_notification: pushResult }, 200);
    }

    // ENDPOINT: /opportunities/decline (POST)
    if (pathname.endsWith("/opportunities/decline") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { match_id, message } = body;

      if (!match_id) {
        return jsonResponse({ success: false, error: "MISSING_MATCH_ID", message: "match_id is required." }, 400);
      }

      return invokeRpc(supabase, "decline_partner_opportunity_secure", {
        p_partner_id: partnerId,
        p_match_id: match_id,
        p_message: message || null,
      });
    }

    // ENDPOINT: /opportunities/propose (POST)
    if (pathname.endsWith("/opportunities/propose") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { match_id, proposed_start_at, proposed_end_at, message } = body;

      if (!match_id || !proposed_start_at || !proposed_end_at || !message) {
        return jsonResponse({
          success: false,
          error: "MISSING_FIELDS",
          message: "match_id, proposed_start_at, proposed_end_at, and message are required.",
        }, 400);
      }

      const { data, error } = await supabase.rpc("propose_partner_alternative_secure", {
        p_partner_id: partnerId,
        p_match_id: match_id,
        p_proposed_start: proposed_start_at,
        p_proposed_end: proposed_end_at,
        p_message: message,
      });

      if (error) {
        return jsonResponse({ success: false, error: "DATABASE_ERROR", message: error.message }, 500);
      }

      if (!data?.success) {
        return jsonResponse(data, 400);
      }

      const pushResult = await dispatchVisitorProposalPushByMatch(match_id, data.response_id);
      return jsonResponse({ ...data, push_notification: pushResult }, 200);
    }

    // ENDPOINT: /opportunities/accept-counter (POST)
    if (pathname.endsWith("/opportunities/accept-counter") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { match_id } = body;

      if (!match_id) {
        return jsonResponse({ success: false, error: "MISSING_MATCH_ID", message: "match_id is required." }, 400);
      }

      return invokeRpc(supabase, "accept_partner_counter_offer_secure", {
        p_partner_id: partnerId,
        p_match_id: match_id,
      });
    }

    // ENDPOINT: /opportunities/decline-counter (POST)
    if (pathname.endsWith("/opportunities/decline-counter") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { match_id, reason } = body;

      if (!match_id) {
        return jsonResponse({ success: false, error: "MISSING_MATCH_ID", message: "match_id is required." }, 400);
      }

      return invokeRpc(supabase, "decline_partner_counter_offer_secure", {
        p_partner_id: partnerId,
        p_match_id: match_id,
        p_reason: reason || null,
      });
    }

    // ENDPOINT: /opportunities/withdraw (POST)
    if (pathname.endsWith("/opportunities/withdraw") && req.method === "POST") {
      const body = await req.json().catch(() => ({}));
      const { match_id, reason } = body;

      if (!match_id) {
        return jsonResponse({ success: false, error: "MISSING_MATCH_ID", message: "match_id is required." }, 400);
      }

      return invokeRpc(supabase, "withdraw_partner_opportunity_secure", {
        p_partner_id: partnerId,
        p_match_id: match_id,
        p_reason: reason || null,
      });
    }

    return jsonResponse({ success: false, error: "NOT_FOUND", message: `Endpoint '${pathname}' not found.` }, 404);
  } catch (err: any) {
    return jsonResponse({ success: false, error: "SERVER_ERROR", message: err.message || "Internal server error." }, 500);
  }
});
